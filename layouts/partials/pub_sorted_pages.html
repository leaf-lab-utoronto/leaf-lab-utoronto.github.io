{{/* Pages sorted by year (desc), then accepted papers before preprints, then date (desc). */}} {{- $keyed := slice -}} {{- range . -}} {{- $accepted := 1 -}} {{- if in (.Params.publication_types | default slice) "preprint" }}{{ $accepted = 0 }}{{ end -}} {{- $key := printf "%s%d%s" (.Date.Format "2006") $accepted (.Date.Format "20060102150405") -}} {{- $keyed = $keyed | append (dict "key" $key "page" .) -}} {{- end -}} {{- $out := slice -}} {{- range sort $keyed "key" "desc" -}} {{- $out = $out | append .page -}} {{- end -}} {{- return $out -}}
