{{/* Archive order: publication year descending, preprints first, then date descending. */}} {{- $keyed := slice -}} {{- range sort . "Title" "asc" -}} {{- $date := partial "publication_date.html" . -}} {{- $preprint := 0 -}} {{- if in (.Params.publication_types | default slice) "preprint" }}{{ $preprint = 1 }}{{ end -}} {{- $key := printf "%s%d%s" (.Date.Format "2006") $preprint ($date.Format "20060102150405") -}} {{- $keyed = $keyed | append (dict "key" $key "page" .) -}} {{- end -}} {{- $out := slice -}} {{- range sort $keyed "key" "desc" -}} {{- $out = $out | append .page -}} {{- end -}} {{- return $out -}}
